theory RQv2_Party_Admission

begin

// Function signature and definition of the equational theory E

functions: fst/1, pair/2, snd/1
equations: fst(<x.1, x.2>) = x.1, snd(<x.1, x.2>) = x.2

restriction Inequality:
  "∀ x #i. (Neq( x, x ) @ #i) ⇒ (⊥)"
  // safety formula

rule (modulo E) CreateParty:
   [ Fr( ~a ) ] --> [ !Party( ~a ) ]

rule (modulo E) SendMessage:
   [ !Party( A ), Fr( ~sid ), Fr( ~m ) ]
  --[ Send( A, ~sid, ~m ) ]->
   [ Out( <A, ~sid, ~m> ), !Sent( A, ~sid, ~m ) ]

rule (modulo E) CreateBatch:
   [ Fr( ~bid ), Fr( ~rst ) ] --> [ OpenSlot1( ~bid, ~rst ) ]

rule (modulo E) CollectSlot1:
   [ OpenSlot1( bid, rst ), In( <A, sid, m> ) ]
  -->
   [ OpenSlot2( bid, rst, A, sid, m ) ]

rule (modulo E) CollectSlot2:
   [ OpenSlot2( bid, rst, A1, sid1, m1 ), In( <A2, sid2, m2> ) ]
  -->
   [ Collected( bid, rst, A1, sid1, m1, A2, sid2, m2 ) ]

rule (modulo E) RejectRepeatedParty:
   [ Collected( bid, rst, A, sid1, m1, A, sid2, m2 ) ]
  --[ Reject( bid, rst ) ]->
   [ BatchRejected( bid, rst ) ]

rule (modulo E) AdmitDistinctParties:
   [ Collected( bid, rst, A1, sid1, m1, A2, sid2, m2 ) ]
  --[ Neq( A1, A2 ), BatchReceive( bid, rst ) ]->
   [ AdmittedSlot1( bid, rst, A1, sid1, m1, A2, sid2, m2 ) ]

rule (modulo E) ProcessSlot1:
   [
   AdmittedSlot1( bid, rst, A1, sid1, m1, A2, sid2, m2 ),
   !Sent( A1, sid1, m1 )
   ]
  --[ ReceiverAccept( A1, sid1, m1, bid, rst ) ]->
   [ AdmittedSlot2( bid, rst, A2, sid2, m2 ) ]

rule (modulo E) ProcessSlot2:
   [ AdmittedSlot2( bid, rst, A2, sid2, m2 ), !Sent( A2, sid2, m2 ) ]
  --[ ReceiverAccept( A2, sid2, m2, bid, rst ) ]->
   [ BatchComplete( bid, rst ) ]

lemma same_party_rejection_exists:
  exists-trace
  "∃ A sid1 sid2 m1 m2 bid rst #s1 #s2 #rj.
    ((((((Send( A, sid1, m1 ) @ #s1) ∧ (Send( A, sid2, m2 ) @ #s2)) ∧
        (Reject( bid, rst ) @ #rj)) ∧
       (¬(sid1 = sid2))) ∧
      (¬(m1 = m2))) ∧
     (#s1 < #rj)) ∧
    (#s2 < #rj)"
/*
guarded formula characterizing all satisfying traces:
"∃ A sid1 sid2 m1 m2 bid rst #s1 #s2 #rj.
  (Send( A, sid1, m1 ) @ #s1) ∧
  (Send( A, sid2, m2 ) @ #s2) ∧
  (Reject( bid, rst ) @ #rj)
 ∧
  (¬(sid1 = sid2)) ∧ (¬(m1 = m2)) ∧ (#s1 < #rj) ∧ (#s2 < #rj)"
*/
by sorry

lemma distinct_party_batch_exists:
  exists-trace
  "∃ A1 A2 sid1 sid2 m1 m2 bid rst #s1 #s2 #b #r1 #r2.
    (((((((((Send( A1, sid1, m1 ) @ #s1) ∧ (Send( A2, sid2, m2 ) @ #s2)) ∧
           (BatchReceive( bid, rst ) @ #b)) ∧
          (ReceiverAccept( A1, sid1, m1, bid, rst ) @ #r1)) ∧
         (ReceiverAccept( A2, sid2, m2, bid, rst ) @ #r2)) ∧
        (¬(A1 = A2))) ∧
       (#s1 < #b)) ∧
      (#s2 < #b)) ∧
     (#b < #r1)) ∧
    (#r1 < #r2)"
/*
guarded formula characterizing all satisfying traces:
"∃ A1 A2 sid1 sid2 m1 m2 bid rst #s1 #s2 #b #r1 #r2.
  (Send( A1, sid1, m1 ) @ #s1) ∧
  (Send( A2, sid2, m2 ) @ #s2) ∧
  (BatchReceive( bid, rst ) @ #b) ∧
  (ReceiverAccept( A1, sid1, m1, bid, rst ) @ #r1) ∧
  (ReceiverAccept( A2, sid2, m2, bid, rst ) @ #r2)
 ∧
  (¬(A1 = A2)) ∧ (#s1 < #b) ∧ (#s2 < #b) ∧ (#b < #r1) ∧ (#r1 < #r2)"
*/
by sorry

lemma accepted_batch_has_distinct_parties:
  all-traces
  "∀ A1 A2 sid1 sid2 m1 m2 bid rst #r1 #r2.
    (((ReceiverAccept( A1, sid1, m1, bid, rst ) @ #r1) ∧
      (ReceiverAccept( A2, sid2, m2, bid, rst ) @ #r2)) ∧
     (¬(#r1 = #r2))) ⇒
    (¬(A1 = A2))"
/*
guarded formula characterizing all counter-examples:
"∃ A1 A2 sid1 sid2 m1 m2 bid rst #r1 #r2.
  (ReceiverAccept( A1, sid1, m1, bid, rst ) @ #r1) ∧
  (ReceiverAccept( A2, sid2, m2, bid, rst ) @ #r2)
 ∧
  (¬(#r1 = #r2)) ∧ (A1 = A2)"
*/
by sorry

lemma receiver_accept_has_send:
  all-traces
  "∀ A sid m bid rst #r.
    (ReceiverAccept( A, sid, m, bid, rst ) @ #r) ⇒
    (∃ #s. (Send( A, sid, m ) @ #s) ∧ (#s < #r))"
/*
guarded formula characterizing all counter-examples:
"∃ A sid m bid rst #r.
  (ReceiverAccept( A, sid, m, bid, rst ) @ #r)
 ∧
  ∀ #s. (Send( A, sid, m ) @ #s) ⇒ ¬(#s < #r)"
*/
by sorry

lemma receiver_accept_injective:
  all-traces
  "∀ A sid m bid rst #s #r1 #r2.
    (((((Send( A, sid, m ) @ #s) ∧
        (ReceiverAccept( A, sid, m, bid, rst ) @ #r1)) ∧
       (ReceiverAccept( A, sid, m, bid, rst ) @ #r2)) ∧
      (#s < #r1)) ∧
     (#s < #r2)) ⇒
    (#r1 = #r2)"
/*
guarded formula characterizing all counter-examples:
"∃ A sid m bid rst #s #r1 #r2.
  (Send( A, sid, m ) @ #s) ∧
  (ReceiverAccept( A, sid, m, bid, rst ) @ #r1) ∧
  (ReceiverAccept( A, sid, m, bid, rst ) @ #r2)
 ∧
  (#s < #r1) ∧ (#s < #r2) ∧ (¬(#r1 = #r2))"
*/
by sorry

end
